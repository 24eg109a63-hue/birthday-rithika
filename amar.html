  <script>

    // Confetti

    (function(){

      const cv = document.getElementById('confetti');

      const ctx = cv.getContext('2d');

      let W, H, pieces = [], running = false, frame;

      function resize(){ W = cv.width = window.innerWidth; H = cv.height = window.innerHeight; }

      window.addEventListener('resize', resize); resize();

      function piece(){

        const colors = ['#ff9bc8','#ffb3d6','#ffc2df','#ffd0e7','#ffe3f1','#ffffff'];

        return { x: Math.random()*W, y: -20, r: 6 + Math.random()*8, c: colors[(Math.random()*colors.length)|0],

                 s: 2 + Math.random()*2.5, a: Math.random()*360, va: -6 + Math.random()*12 }

      }

      function burst(count=150){

        for(let i=0;i<count;i++) pieces.push(piece());

        if(!running){ running = true; loop(); setTimeout(()=>{running=false}, 1800); }

      }

      function loop(){

        if(!running){ cancelAnimationFrame(frame); return }

        frame = requestAnimationFrame(loop);

        ctx.clearRect(0,0,W,H);

        for(let i=0;i<pieces.length;i++){

          const p = pieces[i];

          p.y += p.s; p.x += Math.sin((p.y + i)/20); p.a += p.va;

          if(p.y > H+30){ pieces.splice(i,1); i--; continue }

          ctx.save(); ctx.translate(p.x, p.y); ctx.rotate(p.a * Math.PI/180);

          ctx.fillStyle = p.c; ctx.fillRect(-p.r/2, -p.r/2, p.r, p.r*0.6); ctx.restore();

        }

      }

      window._confettiBurst = burst;

    })();


    // Wishes

    const btn = document.getElementById('wishBtn');

    const list = document.getElementById('wishes');

    btn.addEventListener('click', () => {

      window._confettiBurst(180);

      const who = prompt("Your name?");

      if(who === null) return;

      const msg = prompt("Write your birthday wish for Rithika 💗");

      if(msg === null || !msg.trim()) return;

      const div = document.createElement('div');

      div.className = 'wish';

      div.innerHTML = `<span class="heart">💗</span><div><strong>${escapeHTML(who.trim())}:</strong><div>${escapeHTML(msg.trim())}</div></div>`;

      list.prepend(div);

    });

    function escapeHTML(str){ return str.replace(/[&<>"']/g, s => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[s])); }


    // Song toggle

    function toggleSong(){

      const song = document.getElementById('song');

      if(song.paused){ song.play(); } else { song.pause(); }

    }


    // Sparkle cursor

    document.addEventListener('mousemove', e => {

      el.textContent = ['✨','💖','🌸'][Math.floor(Math.random()*3)];

      document.body.appendChild(el);

      setTimeout(()=>el.remove(), 1000);

    });
